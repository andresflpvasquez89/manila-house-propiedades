function renderFeatured() {
  const grid = document.getElementById('bentoGrid');
  // Pick 4 featured for current mode
  let featured = getCurrentProperties().filter(p => p.featured);
  if (featured.length < 4) {
    // Top by price for monthly which doesn't have featured tag
    featured = [...getCurrentProperties()]
      .sort((a, b) => b.price - a.price)
      .slice(0, 4);
  }
  featured = featured.slice(0, 4);

  const sizes = ['large', 'medium', 'small', 'small'];

  grid.innerHTML = featured.map((p, i) => {
    const isShort = mode === 'short';
    const size = sizes[i] || 'small';
    const priceDisplay = isShort ? formatPrice(p.price, 'USD') : formatPrice(p.price, 'COP');
    const periodDisplay = isShort ? (lang === 'es' ? '/noche' : '/night') : (lang === 'es' ? '/mes' : '/mo');

    const stats = [];
    if (isShort && p.guests) {
      stats.push(`<div class="bento-card-stat"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M17 21v-2a4 4 0 0 0-4-4H5a4 4 0 0 0-4 4v2"/><circle cx="9" cy="7" r="4"/></svg>${p.guests}</div>`);
    }
    if (p.rooms) stats.push(`<div class="bento-card-stat"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M3 9l9-7 9 7v11a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2z"/></svg>${p.rooms}</div>`);
    if (p.bathrooms) stats.push(`<div class="bento-card-stat"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M22 12H2"/><path d="M5.45 5.11L2 12v6a2 2 0 0 0 2 2h16a2 2 0 0 0 2-2v-6l-3.45-6.89A2 2 0 0 0 16.76 4H7.24a2 2 0 0 0-1.79 1.11z"/></svg>${p.bathrooms}</div>`);

    const badge = p.featured
      ? `<span class="bento-card-badge gold">${lang === 'es' ? 'Destacado' : 'Featured'}</span>`
      : (p.rating ? `<span class="bento-card-badge">â˜… ${p.rating}</span>` : '');

    const imgHtml = p.img
      ? `<div class="bento-card-img" style="background-image:url('${p.img}')"></div>`
      : generateBentoPlaceholder(p.name, p.zone);

    return `
      <div class="bento-card ${size}" onclick="openModal('${p.id}')">
        ${imgHtml}
        <div class="bento-card-overlay"></div>
        ${badge}
        <button class="bento-card-fav" onclick="event.stopPropagation();this.classList.toggle('liked');this.textContent=this.classList.contains('liked')?'â™¥':'â™¡'" aria-label="Favorite">â™¡</button>
        <div class="bento-card-info">
          <div class="bento-card-zone">${p.zone}</div>
          <div class="bento-card-name">${p.name}</div>
          <div class="bento-card-meta">
            <div class="bento-card-price num">${priceDisplay}<small>${periodDisplay}</small></div>
            <div class="bento-card-stats">${stats.join('')}</div>
          </div>
        </div>
      </div>
    `;
  }).join('');
}
