function getFiltered() {
  let result = [...getCurrentProperties()];
  const zone = document.getElementById('filterZone').value;
  const type = document.getElementById('filterType').value;
  const sort = document.getElementById('filterSort').value;

  if (zone !== 'all') result = result.filter(p => p.zone === zone);
  if (type !== 'all') result = result.filter(p => p.type === type);
  if (mode === 'short') {
    const guests = parseInt(document.getElementById('filterGuests').value);
    if (guests > 0) result = result.filter(p => (p.guests || 0) >= guests);
  } else {
    const rooms = parseInt(document.getElementById('filterRooms').value);
    if (rooms > 0) result = result.filter(p => (p.rooms || 0) >= rooms);
  }

  if (sort === 'price-low') result.sort((a, b) => a.price - b.price);
  else if (sort === 'price-high') result.sort((a, b) => b.price - a.price);
  else if (sort === 'capacity') result.sort((a, b) => (b.guests || b.rooms || 0) - (a.guests || a.rooms || 0));
  else result.sort((a, b) => ((b.featured ? 1 : 0) - (a.featured ? 1 : 0)) || (b.price - a.price));

  return result;
}

  currentPage = 1;
  renderCollection();
}
